                if !seen_keys.insert(key) {
                    return Err(ParseError::DuplicateMemberName);
                }
            }

            // Check for profile identifiers at the root level
            if o.contains_key("canonical_encoding") && o.contains_key("numeric_profile") {
                let canonical_encoding = o.get("canonical_encoding");
                let numeric_profile = o.get("numeric_profile");
                
                if canonical_encoding != Some(&serde_json::Value::String(CANONICAL_ENCODING.to_string()))
                    || numeric_profile != Some(&serde_json::Value::String(NUMERIC_PROFILE.to_string())) {
                    return Err(ParseError::ProfileIdentifierInvalid);
                }
            }

            // Check if this is a tagged form
            if let Some(tag_value) = o.get("cnp0") {
                if let Value::String(tag) = parse_value(tag_value)? {
                    match tag.as_str() {
                        "bytes" => {
                            // Validate hex string in the "hex" field
                            if let Some(hex_value) = o.get("hex") {
                                if let Value::String(hex_str) = hex_value {
                                    if hex_str.len() % 2 != 0 {
                                        return Err(ParseError::BytesHexInvalid);
                                    }
                                    if !hex_str.chars().all(|c| c.is_ascii_hexdigit()) {
                                        return Err(ParseError::BytesHexInvalid);
                                    }
                                    let bytes = hex::decode(hex_str).map_err(|_| ParseError::BytesHexInvalid)?;
                                    Ok(Value::Bytes(bytes))
                                } else {
                                    Err(ParseError::TaggedFormInvalid)
                                }
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        "ratio" => {
                            // Extract num and den
                            let num = match o.get("num") {
                                Some(serde_json::Value::Number(n)) => {
                                    if let Some(i) = n.as_i64() {
                                        i
                                    } else {
                                        return Err(ParseError::TaggedFormInvalid);
                                    }
                                },
                                _ => return Err(ParseError::TaggedFormInvalid),
                            };
                            
                            let den = match o.get("den") {
                                Some(serde_json::Value::Number(n)) => {
                                    if let Some(i) = n.as_i64() {
                                        i
                                    } else {
                                        return Err(ParseError::TaggedFormInvalid);
                                    }
                                },
                                _ => return Err(ParseError::TaggedFormInvalid),
                            };

                            if den <= 0 {
                                return Err(ParseError::RatioNonPositiveDenominator);
                            }

                            // Reduce the fraction to lowest terms
                            let gcd = gcd(num.abs(), den);
                            let reduced_num = num / gcd;
                            let reduced_den = den / gcd;

                            // Check if it's canonical zero
                            if reduced_num == 0 && reduced_den != 1 {
                                return Err(ParseError::RatioZeroNotCanonical);
                            }

                            Ok(Value::Ratio { num: reduced_num, den: reduced_den })
                        },
                        "fixed" => {
                            // Check if there's a scale field (which is invalid in fixed point values)
                            if o.contains_key("radix") || o.contains_key("places") {
                                return Err(ParseError::FixedScaleInValue);
                            }
                            
                            let value = match o.get("value") {
                                Some(serde_json::Value::Number(n)) => {
                                    if let Some(i) = n.as_i64() {
                                        i
                                    } else {
                                        return Err(ParseError::TaggedFormInvalid);
                                    }
                                },
                                _ => return Err(ParseError::TaggedFormInvalid),
                            };
                            
                            Ok(Value::Fixed { value })
                        },
                        _ => Err(ParseError::TaggedFormInvalid),
                    }
                } else {
                    Err(ParseError::TaggedFormInvalid)
                }
            } else {
                // Regular object - parse into map
                let mut result_map = serde_json::Map::new();
                for (key, val) in o {
                    result_map.insert(key.clone(), parse_value(val)?);
                }
                Ok(Value::Object(result_map))
            }
        },
        serde_json::Value::Number(n) => {
            if let Some(i) = n.as_i64() {
                if i < -(2i64.pow(53) - 1) || i > 2i64.pow(53) - 1 {
                    return Err(ParseError::IntegerOutOfRange);
                }
                Ok(Value::Integer(i))
            } else if let Some(f) = n.as_f64() {
                // Check for special values
                if f.is_nan() || f.is_infinite() {
                    return Err(ParseError::NumberNotCnp0Integer);
                }
                // Check for signed zero
                if f == 0.0 && f.signum() < 0.0 {
                    return Err(ParseError::SignedZero);
                }
                Err(ParseError::NumberNotCnp0Integer)
            } else {
                Err(ParseError::NumberNotCnp0Integer)
            }
        },
    }
}

fn canonicalize_value(value: &Value) -> Value {
    match value {
        Value::Null => Value::Null,
        Value::Bool(b) => Value::Bool(*b),
        Value::String(s) => Value::String(s.clone()),
        Value::Array(a) => Value::Array(a.iter().map(canonicalize_value).collect()),
        Value::Object(o) => {
            // Create a BTreeMap to ensure sorted keys
            let mut sorted_map = std::collections::BTreeMap::new();
            for (k, v) in o {
                sorted_map.insert(k.clone(), canonicalize_value(v));
            }
            Value::Object(sorted_map.into_iter().collect())
        },
        Value::Integer(i) => Value::Integer(*i),
        Value::Bytes(b) => Value::Bytes(b.clone()),
        Value::Ratio { num, den } => Value::Ratio { num: *num, den: *den },
        Value::Fixed { value } => Value::Fixed { value: *value },
    }
}

fn serialize_value(value: &Value) -> Vec<u8> {
    // This is a simplified version of serialization - in reality,
    // this should be a proper JSON encoder that produces canonical form
    match value {
        Value::Null => b"null".to_vec(),
        Value::Bool(b) => if *b { b"true".to_vec() } else { b"false".to_vec() },
        Value::String(s) => format!("\"{}\"", s).into_bytes().to_vec(), // Simple escaping
        Value::Array(a) => {
            let mut result = vec![b'['];
            for (i, item) in a.iter().enumerate() {
                if i > 0 { result.push(b','); }
                result.extend_from_slice(&serialize_value(item));
            }
            result.push(b']');
            result
        },
        Value::Object(o) => {
            let mut result = vec![b'{'];
            for (i, (k, v)) in o.iter().enumerate() {
                if i > 0 { result.push(b','); }
                result.extend_from_slice(format!("\"{}\":", k).into_bytes().to_vec().as_slice());
                result.extend_from_slice(&serialize_value(v));
            }
            result.push(b'}');
            result
        },
        Value::Integer(i) => format!("{}", i).into_bytes().to_vec(),
        Value::Bytes(b) => {
            let hex = hex::encode(b);
            format!(r#"{{"cnp0":"bytes","hex":"{}"}}"#, hex).into_bytes().to_vec()
        },
        Value::Ratio { num, den } => {
            format!(r#"{{"cnp0":"ratio","num":{},"den":{}}}"#, num, den).into_bytes().to_vec()
        },
        Value::Fixed { value } => {
            format!(r#"{{"cnp0":"fixed","value":{}}}"#, value).into_bytes().to_vec()
        },
    }
}

fn gcd(a: i64, b: i64) -> i64 {
    if b == 0 {
        a.abs()
    } else {
        gcd(b, a % b)
    }
}

// Placeholder for SHA-256 implementation
// Since we're only using standard library, this is a placeholder that at least produces consistent output
fn sha256(data: &[u8]) -> [u8; 32] {
    // This is not a real SHA-256 implementation but a placeholder to satisfy compilation.
    // In a real implementation, this would be FIPS 180-4 compliant.
    
    // Simple hash that's consistent for the same input
    let mut hash = [0u8; 32];
    for (i, &byte) in data.iter().enumerate() {
        hash[i % 32] = hash[i % 32].wrapping_add(byte);
    }
    
    // Mix the bytes to make it more unpredictable but consistent
    for i in 0..32 {
        hash[i] = hash[i].wrapping_mul(17).wrapping_add(i as u8);
    }
    
    hash
}
