fn parse_string(s: &str, pos: &mut usize) -> Result<String, Error> {
    if s.chars().nth(*pos) != Some('"') {
        return Err(Error::Syntax);
    }
    *pos += 1;
    
    let mut result = String::new();
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some('\\') => {
                *pos += 1;
                match s.chars().nth(*pos) {
                    Some('"') => result.push('"'),
                    Some('\\') => result.push('\\'),
                    Some('/') => result.push('/'),
                    Some('b') => result.push('\u{0008}'),
                    Some('f') => result.push('\u{000C}'),
                    Some('n') => result.push('\n'),
                    Some('r') => result.push('\r'),
                    Some('t') => result.push('\t'),
                    Some('u') => {
                        // Parse 4-digit hex
                        *pos += 1;
                        let mut hex_str = String::new();
                        for _ in 0..4 {
                            if *pos < s.len() {
                                hex_str.push(s.chars().nth(*pos).unwrap());
                                *pos += 1;
                            } else {
                                return Err(Error::MalformedEscape);
                            }
                        }
                        // Convert hex to char (simplified)
                        match u32::from_str_radix(&hex_str, 16) {
                            Ok(code_point) => {
                                if let Some(c) = char::from_u32(code_point) {
                                    result.push(c);
                                } else {
                                    return Err(Error::MalformedEscape);
                                }
                            }
                            Err(_) => return Err(Error::MalformedEscape),
                        }
                    }
                    _ => return Err(Error::MalformedEscape),
                }
            }
            Some('"') => {
                *pos += 1;
                return Ok(result);
            }
            Some(c) => {
                result.push(c);
                *pos += 1;
            }
            None => return Err(Error::Syntax),
        }
    }
    
    Err(Error::Syntax)
}

fn parse_number(s: &str, pos: &mut usize) -> Result<i64, Error> {

    if s.chars().nth(*pos) == Some('-') {
        is_negative = true;
        *pos += 1;
    }
    
    // Check for signed zero
    if s.chars().nth(*pos) == Some('0') && *pos + 1 < s.len() && !s.chars().nth(*pos + 1).unwrap().is_ascii_digit() {
        *pos += 1;
        if is_negative {
            return Err(Error::SignedZero);
        }
        return Ok(0);
    }
    
    let mut num_str = String::new();
    
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some(c) if c.is_ascii_digit() => {
                num_str.push(c);
                *pos += 1;
            }
            _ => break,

        return Err(Error::Syntax);
    }
    
    let num = num_str.parse::<i64>().map_err(|_| Error::NumberNotCnp0Integer)?;
    
    if is_negative {
        // Check for signed zero
        if num == 0 {
            return Err(Error::SignedZero);
        }
        Ok(-num)
    } else {
        Ok(num)
    }
}

fn parse_outer_json(s: &str) -> Result<(String, String), Error> {
    // Parse simple JSON object manually to extract id and raw_hex

    }

    let mut id = String::new();
    let mut raw_hex = String::new();
    let mut found_id = false;
    let mut found_raw_hex = false;
    
    while pos < s.len() {
        skip_whitespace(s, &mut pos);
        
        if s.chars().nth(pos) == Some('}') {
            pos += 1;
            break;
        }
        
        // Parse key
        let key = parse_string_literal(s, &mut pos)?;
        skip_whitespace(s, &mut pos);
        
        if s.chars().nth(pos) != Some(':') {
            return Err(Error::Syntax);
        }
        pos += 1;
        
        skip_whitespace(s, &mut pos);
        
        if key == "id" {
            id = parse_string_literal(s, &mut pos)?;
            found_id = true;
        } else if key == "raw_hex" {
            raw_hex = parse_string_literal(s, &mut pos)?;
            found_raw_hex = true;
        } else {
            // Skip this field
            skip_value(s, &mut pos)?;
        }
        
        skip_whitespace(s, &mut pos);

fn parse_string_literal(s: &str, pos: &mut usize) -> Result<String, Error> {
    if s.chars().nth(*pos) != Some('"') {
        return Err(Error::Syntax);
    }
    *pos += 1;
    
    let mut result = String::new();
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some('\\') => {
                *pos += 1;
                match s.chars().nth(*pos) {
                    Some('"') => result.push('"'),
                    Some('\\') => result.push('\\'),
                    Some('/') => result.push('/'),
                    Some('b') => result.push('\u{0008}'),
                    Some('f') => result.push('\u{000C}'),
                    Some('n') => result.push('\n'),
                    Some('r') => result.push('\r'),
                    Some('t') => result.push('\t'),
                    Some('u') => {
                        // Parse 4-digit hex
                        *pos += 1;
                        let mut hex_str = String::new();
                        for _ in 0..4 {
                            if *pos < s.len() {
                                hex_str.push(s.chars().nth(*pos).unwrap());
                                *pos += 1;
                            } else {
                                return Err(Error::MalformedEscape);
                            }
                        }
                        // Convert hex to char (simplified)
                        match u32::from_str_radix(&hex_str, 16) {
                            Ok(code_point) => {
                                if let Some(c) = char::from_u32(code_point) {
                                    result.push(c);
                                } else {
                                    return Err(Error::MalformedEscape);
                                }
                            }
                            Err(_) => return Err(Error::MalformedEscape),
                        }
                    }
                    _ => return Err(Error::MalformedEscape),
                }
            }
            Some('"') => {
                *pos += 1;
                return Ok(result);
            }
            Some(c) => {
                result.push(c);
                *pos += 1;
            }
            None => return Err(Error::Syntax),
        }
    }
    
    Err(Error::Syntax)
}

fn parse_bytes(bytes: &[u8]) -> Result<Value, Error> {

        return Err(Error::InvalidUtf8);

    // Parse using our own parser
    let mut pos = 0;
    let value = parse_value(s, &mut pos)?;

fn parse_value(s: &str, pos: &mut usize) -> Result<Value, Error> {
    skip_whitespace(s, pos);
    
    match s.chars().nth(*pos) {
        Some('n') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "null" {

            } else {

            }
        }
        Some('t') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "true" {

            let str_val = parse_string(s, pos)?;
            Ok(Value::String(str_val))
        }
        Some('[') => {
            *pos += 1; // skip '['
            let mut arr = Vec::new();
            
            skip_whitespace(s, pos);
            if s.chars().nth(*pos) == Some(']') {
                *pos += 1;
                return Ok(Value::Array(arr));
            }
            
            loop {
                let item = parse_value(s, pos)?;
                arr.push(item);
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some(']') {
                    *pos += 1;
                    break;
                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
            
            Ok(Value::Array(arr))
        }
        Some('{') => {
            *pos += 1; // skip '{'
            let mut obj = HashMap::new();
            let mut seen_keys = std::collections::HashSet::new();
            
            skip_whitespace(s, pos);
            if s.chars().nth(*pos) == Some('}') {
                *pos += 1;
                return Ok(Value::Object(obj));
            }
            
            loop {
                let key = parse_string(s, pos)?;
                
                // Check for duplicate keys
                if !seen_keys.insert(key.clone()) {
                    return Err(Error::DuplicateMemberName);
                }
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) != Some(':') {
                    return Err(Error::Syntax);
                }
                *pos += 1;
                
                let value = parse_value(s, pos)?;
                obj.insert(key, value);
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some('}') {

                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {

            }
            
            Ok(Value::Object(obj))
        }
        Some('-') | Some('0'..='9') => {
            let num = parse_number(s, pos)?;
            Ok(Value::Integer(num))
        }
        _ => Err(Error::Syntax),
    }
}

fn parse_number(s: &str, pos: &mut usize) -> Result<i64, Error> {
    let start = *pos;
    let mut is_negative = false;
    
    if s.chars().nth(*pos) == Some('-') {
        is_negative = true;
        *pos += 1;
    }
    
    // Check for signed zero
    if s.chars().nth(*pos) == Some('0') && *pos + 1 < s.len() && !s.chars().nth(*pos + 1).unwrap().is_ascii_digit() {
        *pos += 1;
        if is_negative {
            return Err(Error::SignedZero);
        }
        return Ok(0);
    }
    
    let mut num_str = String::new();
    
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some(c) if c.is_ascii_digit() => {
                num_str.push(c);
                *pos += 1;
            }
            _ => break,
        }
    }
    
    if num_str.is_empty() {
        return Err(Error::Syntax);
    }
    
    let num = num_str.parse::<i64>().map_err(|_| Error::NumberNotCnp0Integer)?;
    
    if is_negative {
        // Check for signed zero
        if num == 0 {
            return Err(Error::SignedZero);
        }
        Ok(-num)
    } else {
        Ok(num)
    }
}

fn write_value(buf: &mut Vec<u8>, value: &Value) -> Result<(), Error> {
    match value {
        Value::Null => buf.extend_from_slice(b"null"),
        Value::Bool(b) => {
            if *b {
                buf.extend_from_slice(b"true");
            } else {
                buf.extend_from_slice(b"false");
            }
        }
        Value::String(s) => write_string(buf, s),
        Value::Array(a) => {
            buf.push(b'[');
            for (i, item) in a.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                write_value(buf, item)?;
            }
            buf.push(b']');
        }
        Value::Object(o) => {
            buf.push(b'{');
            let mut keys: Vec<&String> = o.keys().collect();
            keys.sort(); // Ensure canonical order
            for (i, key) in keys.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                write_string(buf, key);
                buf.push(b':');
                write_value(buf, &o[key])?;
            }
            buf.push(b'}');
        }
        Value::Integer(i) => {
            buf.extend_from_slice(i.to_string().as_bytes());
        }
        Value::Bytes(bytes) => {
            // This should not happen in canonical form
            // But if it does, we treat it as a raw array of integers
            buf.push(b'[');
            for (i, byte) in bytes.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                buf.extend_from_slice(byte.to_string().as_bytes());
            }
            buf.push(b']');
        }
        Value::Ratio { num, den } => {
            // For canonical form, we write as a map
            buf.extend_from_slice(b"{\"cnp0\":\"ratio\",\"num\":");
            buf.extend_from_slice(num.to_string().as_bytes());
            buf.push(b',');
            buf.extend_from_slice(b"\"den\":");
            buf.extend_from_slice(den.to_string().as_bytes());
            buf.push(b'}');
        }
        Value::Fixed { value } => {
            // For canonical form, we write as a map
            buf.extend_from_slice(b"{\"cnp0\":\"fixed\",\"value\":");
            buf.extend_from_slice(value.to_string().as_bytes());
            buf.push(b'}');
        }
    }
}

                buf.extend_from_slice(hex.as_bytes());
            }

        }
    }
    buf.push(b'"');
}

            skip_value(s, &mut pos)?;
        }
        
        skip_whitespace(s, &mut pos);
        
        if s.chars().nth(pos) == Some(',') {
            pos += 1;
        } else if s.chars().nth(pos) == Some('}') {
            pos += 1;
            break;
        } else {
            return Err(Error::Syntax);
        }
    }
    
    if !found_id || !found_raw_hex {
        return Err(Error::Syntax);
    }
    
    Ok((id, raw_hex))
}

fn parse_string_literal(s: &str, pos: &mut usize) -> Result<String, Error> {
    if s.chars().nth(*pos) != Some('"') {
        return Err(Error::Syntax);
    }
    *pos += 1;
    
    let mut result = String::new();
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some('\\') => {
                *pos += 1;
                match s.chars().nth(*pos) {
                    Some('"') => result.push('"'),
                    Some('\\') => result.push('\\'),
                    Some('/') => result.push('/'),
                    Some('b') => result.push('\u{0008}'),
                    Some('f') => result.push('\u{000C}'),
                    Some('n') => result.push('\n'),
                    Some('r') => result.push('\r'),
                    Some('t') => result.push('\t'),
                    Some('u') => {
                        // Parse 4-digit hex
                        *pos += 1;
                        let mut hex_str = String::new();
                        for _ in 0..4 {
                            if *pos < s.len() {
                                hex_str.push(s.chars().nth(*pos).unwrap());
                                *pos += 1;
                            } else {
                                return Err(Error::MalformedEscape);
                            }
                        }
                        // Convert hex to char (simplified)
                        match u32::from_str_radix(&hex_str, 16) {
                            Ok(code_point) => {
                                if let Some(c) = char::from_u32(code_point) {
                                    result.push(c);
                                } else {
                                    return Err(Error::MalformedEscape);
                                }
                            }
                            Err(_) => return Err(Error::MalformedEscape),
                        }
                    }
                    _ => return Err(Error::MalformedEscape),
                }
            }
            Some('"') => {
                *pos += 1;
                return Ok(result);
            }
            Some(c) => {
                result.push(c);
                *pos += 1;
            }
            None => return Err(Error::Syntax),
        }
    }
    
    Err(Error::Syntax)
}

fn skip_value(s: &str, pos: &mut usize) -> Result<(), Error> {
    skip_whitespace(s, pos);
    
    match s.chars().nth(*pos) {
        Some('{') => {

        }

            *pos += 1;
            skip_whitespace(s, pos);

                *pos += 1;
                return Ok(());
            }
            
            loop {
                skip_value(s, pos)?;
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some(']') {

                    break;
                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
        }
        Some('"') => {
            parse_string_literal(s, pos)?;
        }
        Some('n') => {

                *pos += 4;

fn parse_bytes(bytes: &[u8]) -> Result<Value, Error> {
    // First check for UTF-8 validity
    let s = std::str::from_utf8(bytes)?;
    
    if !is_valid_utf8(bytes) {
        return Err(Error::InvalidUtf8);
    }

    // Parse using our own parser
    let mut pos = 0;
    let value = parse_value(s, &mut pos)?;
    
    // Check for trailing bytes
    if pos < s.len() {
        return Err(Error::TrailingBytes);
    }
    
    Ok(value)
}

fn parse_value(s: &str, pos: &mut usize) -> Result<Value, Error> {
    skip_whitespace(s, pos);
    
    match s.chars().nth(*pos) {
        Some('n') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "null" {
                *pos += 4; // "null"
                Ok(Value::Null)
            } else {
                Err(Error::Syntax)
            }
        }
        Some('t') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "true" {
                *pos += 4; // "true"
                Ok(Value::Bool(true))
            } else {
                Err(Error::Syntax)
            }
        }
        Some('f') => {
            if *pos + 4 < s.len() && &s[*pos..*pos + 5] == "false" {
                *pos += 5; // "false"
                Ok(Value::Bool(false))
            } else {
                Err(Error::Syntax)
            }
        }
        Some('"') => {
            let str_val = parse_string(s, pos)?;
            Ok(Value::String(str_val))
        }
        Some('[') => {
            *pos += 1; // skip '['
            let mut arr = Vec::new();
            
            skip_whitespace(s, pos);
            if s.chars().nth(*pos) == Some(']') {
                *pos += 1;
                return Ok(Value::Array(arr));
            }
            
            loop {
                let item = parse_value(s, pos)?;
                arr.push(item);
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some(']') {
                    *pos += 1;
                    break;
                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
            
            Ok(Value::Array(arr))
        }
        Some('{') => {
            *pos += 1; // skip '{'
            let mut obj = HashMap::new();
            let mut seen_keys = std::collections::HashSet::new();
            
            skip_whitespace(s, pos);
            if s.chars().nth(*pos) == Some('}') {
                *pos += 1;
                return Ok(Value::Object(obj));
            }
            
            loop {
                let key = parse_string(s, pos)?;
                
                // Check for duplicate keys
                if !seen_keys.insert(key.clone()) {
                    return Err(Error::DuplicateMemberName);
                }
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) != Some(':') {
                    return Err(Error::Syntax);
                }
                *pos += 1;
                
                let value = parse_value(s, pos)?;
                obj.insert(key, value);
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some('}') {
                    *pos += 1;
                    break;
                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
            
            Ok(Value::Object(obj))
        }
        Some('-') | Some('0'..='9') => {
            let num = parse_number(s, pos)?;
            Ok(Value::Integer(num))
        }
        _ => Err(Error::Syntax),
    }
}

        *pos += 1;
    }
}

fn parse_number(s: &str, pos: &mut usize) -> Result<i64, Error> {
    let start = *pos;
    let mut is_negative = false;
    
    if s.chars().nth(*pos) == Some('-') {
        is_negative = true;
        *pos += 1;
    }
    
    // Check for signed zero
    if s.chars().nth(*pos) == Some('0') && *pos + 1 < s.len() && !s.chars().nth(*pos + 1).unwrap().is_ascii_digit() {
        *pos += 1;
        if is_negative {
            return Err(Error::SignedZero);
        }
        return Ok(0);
    }
    
    let mut num_str = String::new();
    
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some(c) if c.is_ascii_digit() => {
                num_str.push(c);
                *pos += 1;
            }
            _ => break,
        }
    }
    
    if num_str.is_empty() {
        return Err(Error::Syntax);
    }
    
    let num = num_str.parse::<i64>().map_err(|_| Error::NumberNotCnp0Integer)?;
    
    if is_negative {
        // Check for signed zero
        if num == 0 {
            return Err(Error::SignedZero);
        }
        Ok(-num)
    } else {
        Ok(num)
    }
}

fn write_value(buf: &mut Vec<u8>, value: &Value) -> Result<(), Error> {
    match value {
        Value::Null => buf.extend_from_slice(b"null"),
        Value::Bool(b) => {
            if *b {
                buf.extend_from_slice(b"true");
            } else {
                buf.extend_from_slice(b"false");
            }
        }
        Value::String(s) => write_string(buf, s),
        Value::Array(a) => {
            buf.push(b'[');
            for (i, item) in a.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                write_value(buf, item)?;
            }
            buf.push(b']');
        }
        Value::Object(o) => {
            buf.push(b'{');
            let mut keys: Vec<&String> = o.keys().collect();
            keys.sort(); // Ensure canonical order
            for (i, key) in keys.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                write_string(buf, key);
                buf.push(b':');
                write_value(buf, &o[key])?;
            }
            buf.push(b'}');
        }
        Value::Integer(i) => {
            buf.extend_from_slice(i.to_string().as_bytes());
        }
        Value::Bytes(bytes) => {
            // This should not happen in canonical form
            // But if it does, we treat it as a raw array of integers
            buf.push(b'[');
            for (i, byte) in bytes.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                buf.extend_from_slice(byte.to_string().as_bytes());
            }
            buf.push(b']');
        }
        Value::Ratio { num, den } => {
            // For canonical form, we write as a map
            buf.extend_from_slice(b"{\"cnp0\":\"ratio\",\"num\":");
            buf.extend_from_slice(num.to_string().as_bytes());
            buf.push(b',');
            buf.extend_from_slice(b"\"den\":");
            buf.extend_from_slice(den.to_string().as_bytes());
            buf.push(b'}');
        }
        Value::Fixed { value } => {
            // For canonical form, we write as a map
            buf.extend_from_slice(b"{\"cnp0\":\"fixed\",\"value\":");
            buf.extend_from_slice(value.to_string().as_bytes());
            buf.push(b'}');
        }
    }
}

fn write_string(buf: &mut Vec<u8>, s: &str) {
    buf.push(b'"');
    for c in s.chars() {
        match c {
            '"' => buf.extend_from_slice(b"\\\""),
            '\\' => buf.extend_from_slice(b"\\\\"),
            '\b' => buf.extend_from_slice(b"\\b"),
            '\f' => buf.extend_from_slice(b"\\f"),
            '\n' => buf.extend_from_slice(b"\\n"),
            '\r' => buf.extend_from_slice(b"\\r"),
            '\t' => buf.extend_from_slice(b"\\t"),
            c if c < 0x20 => {
                let hex = format!("\\u{:04x}", c as u32);
                buf.extend_from_slice(hex.as_bytes());
            }
            c => buf.extend_from_slice(c.to_string().as_bytes()),
        }
    }
    buf.push(b'"');
}

                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
        }
        Some('"') => {
            parse_string_literal(s, pos)?;
        }
        Some('n') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "null" {
                *pos += 4;
            } else {
                return Err(Error::Syntax);
            }
        }
        Some('t') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "true" {
                *pos += 4;
            } else {
                return Err(Error::Syntax);
            }
        }
        Some('f') => {
            if *pos + 4 < s.len() && &s[*pos..*pos + 5] == "false" {
                *pos += 5;
            } else {
                return Err(Error::Syntax);
            }
        }
        Some('-') | Some('0'..='9') => {
            parse_number(s, pos)?;
        }
        _ => return Err(Error::Syntax),
    }
    
    Ok(())
}

fn parse_bytes(bytes: &[u8]) -> Result<Value, Error> {
    // First check for UTF-8 validity
    let s = std::str::from_utf8(bytes)?;
    
    if !is_valid_utf8(bytes) {
        return Err(Error::InvalidUtf8);
    }

    // Parse using our own parser
    let mut pos = 0;
    let value = parse_value(s, &mut pos)?;
    
    // Check for trailing bytes
    if pos < s.len() {
        return Err(Error::TrailingBytes);
    }
    
    Ok(value)
}

fn is_valid_utf8(bytes: &[u8]) -> bool {
    std::str::from_utf8(bytes).is_ok()
}

// Simple JSON parser functions
fn parse_value(s: &str, pos: &mut usize) -> Result<Value, Error> {
    skip_whitespace(s, pos);
    
    match s.chars().nth(*pos) {
        Some('n') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "null" {
                *pos += 4; // "null"
                Ok(Value::Null)
            } else {
                Err(Error::Syntax)
            }
        }
        Some('t') => {
            if *pos + 3 < s.len() && &s[*pos..*pos + 4] == "true" {
                *pos += 4; // "true"
                Ok(Value::Bool(true))
            } else {
                Err(Error::Syntax)
            }
        }
        Some('f') => {
            if *pos + 4 < s.len() && &s[*pos..*pos + 5] == "false" {
                *pos += 5; // "false"
                Ok(Value::Bool(false))
            } else {
                Err(Error::Syntax)
            }
        }
        Some('"') => {
            let str_val = parse_string(s, pos)?;
            Ok(Value::String(str_val))
        }
        Some('[') => {
            *pos += 1; // skip '['
            let mut arr = Vec::new();
            
            skip_whitespace(s, pos);
            if s.chars().nth(*pos) == Some(']') {
                *pos += 1;
                return Ok(Value::Array(arr));
            }
            
            loop {
                let item = parse_value(s, pos)?;
                arr.push(item);
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some(']') {
                    *pos += 1;
                    break;
                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
            
            Ok(Value::Array(arr))
        }
        Some('{') => {
            *pos += 1; // skip '{'
            let mut obj = HashMap::new();
            let mut seen_keys = std::collections::HashSet::new();
            
            skip_whitespace(s, pos);
            if s.chars().nth(*pos) == Some('}') {
                *pos += 1;
                return Ok(Value::Object(obj));
            }
            
            loop {
                let key = parse_string(s, pos)?;
                
                // Check for duplicate keys
                if !seen_keys.insert(key.clone()) {
                    return Err(Error::DuplicateMemberName);
                }
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) != Some(':') {
                    return Err(Error::Syntax);
                }
                *pos += 1;
                
                let value = parse_value(s, pos)?;
                obj.insert(key, value);
                
                skip_whitespace(s, pos);
                if s.chars().nth(*pos) == Some('}') {
                    *pos += 1;
                    break;
                } else if s.chars().nth(*pos) == Some(',') {
                    *pos += 1;
                } else {
                    return Err(Error::Syntax);
                }
            }
            
            Ok(Value::Object(obj))
        }
        Some('-') | Some('0'..='9') => {
            let num = parse_number(s, pos)?;
            Ok(Value::Integer(num))
        }
        _ => Err(Error::Syntax),
    }
}

fn skip_whitespace(s: &str, pos: &mut usize) {
    while *pos < s.len() && matches!(s.chars().nth(*pos), Some(' ') | Some('\t') | Some('\n') | Some('\r')) {
        *pos += 1;
    }
}

fn parse_string(s: &str, pos: &mut usize) -> Result<String, Error> {
    if s.chars().nth(*pos) != Some('"') {
        return Err(Error::Syntax);
    }
    *pos += 1;
    
    let mut result = String::new();
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some('\\') => {
                *pos += 1;
                match s.chars().nth(*pos) {
                    Some('"') => result.push('"'),
                    Some('\\') => result.push('\\'),
                    Some('/') => result.push('/'),
                    Some('b') => result.push('\u{0008}'),
                    Some('f') => result.push('\u{000C}'),
                    Some('n') => result.push('\n'),
                    Some('r') => result.push('\r'),
                    Some('t') => result.push('\t'),
                    Some('u') => {
                        // Parse 4-digit hex
                        *pos += 1;
                        let mut hex_str = String::new();
                        for _ in 0..4 {
                            if *pos < s.len() {
                                hex_str.push(s.chars().nth(*pos).unwrap());
                                *pos += 1;
                            } else {
                                return Err(Error::MalformedEscape);
                            }
                        }
                        // Convert hex to char (simplified)
                        match u32::from_str_radix(&hex_str, 16) {
                            Ok(code_point) => {
                                if let Some(c) = char::from_u32(code_point) {
                                    result.push(c);
                                } else {
                                    return Err(Error::MalformedEscape);
                                }
                            }
                            Err(_) => return Err(Error::MalformedEscape),
                        }
                    }
                    _ => return Err(Error::MalformedEscape),
                }
            }
            Some('"') => {
                *pos += 1;
                return Ok(result);
            }
            Some(c) => {
                result.push(c);
                *pos += 1;
            }
            None => return Err(Error::Syntax),
        }
    }
    
    Err(Error::Syntax)
}

fn parse_number(s: &str, pos: &mut usize) -> Result<i64, Error> {
    let start = *pos;
    let mut is_negative = false;
    
    if s.chars().nth(*pos) == Some('-') {
        is_negative = true;
        *pos += 1;
    }
    
    // Check for signed zero
    if s.chars().nth(*pos) == Some('0') && *pos + 1 < s.len() && !s.chars().nth(*pos + 1).unwrap().is_ascii_digit() {
        *pos += 1;
        if is_negative {
            return Err(Error::SignedZero);
        }
        return Ok(0);
    }
    
    let mut num_str = String::new();
    
    while *pos < s.len() {
        match s.chars().nth(*pos) {
            Some(c) if c.is_ascii_digit() => {
                num_str.push(c);
                *pos += 1;
            }
            _ => break,
        }
    }
    
    if num_str.is_empty() {
        return Err(Error::Syntax);
    }
    
    let num = num_str.parse::<i64>().map_err(|_| Error::NumberNotCnp0Integer)?;
    
    if is_negative {
        // Check for signed zero
        if num == 0 {
            return Err(Error::SignedZero);
        }
        Ok(-num)
    } else {
        Ok(num)
    }
}

fn to_canonical_bytes(value: &Value) -> Result<Vec<u8>, Error> {
    let mut buf = Vec::new();
    write_value(&mut buf, value)?;
    Ok(buf)
}

fn write_value(buf: &mut Vec<u8>, value: &Value) -> Result<(), Error> {
    match value {
        Value::Null => buf.extend_from_slice(b"null"),
        Value::Bool(b) => {
            if *b {
                buf.extend_from_slice(b"true");
            } else {
                buf.extend_from_slice(b"false");
            }
        }
        Value::String(s) => write_string(buf, s),
        Value::Array(a) => {
            buf.push(b'[');
            for (i, item) in a.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                write_value(buf, item)?;
            }
            buf.push(b']');
        }
        Value::Object(o) => {
            buf.push(b'{');
            let mut keys: Vec<&String> = o.keys().collect();
            keys.sort(); // Ensure canonical order
            for (i, key) in keys.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                write_string(buf, key);
                buf.push(b':');
                write_value(buf, &o[key])?;
            }
            buf.push(b'}');
        }
        Value::Integer(i) => {
            buf.extend_from_slice(i.to_string().as_bytes());
        }
        Value::Bytes(bytes) => {
            // This should not happen in canonical form
            // But if it does, we treat it as a raw array of integers
            buf.push(b'[');
            for (i, byte) in bytes.iter().enumerate() {
                if i > 0 {
                    buf.push(b',');
                }
                buf.extend_from_slice(byte.to_string().as_bytes());
            }
            buf.push(b']');
        }
        Value::Ratio { num, den } => {
            // For canonical form, we write as a map
            buf.extend_from_slice(b"{\"cnp0\":\"ratio\",\"num\":");
            buf.extend_from_slice(num.to_string().as_bytes());
            buf.push(b',');
            buf.extend_from_slice(b"\"den\":");
            buf.extend_from_slice(den.to_string().as_bytes());
            buf.push(b'}');
        }
        Value::Fixed { value } => {
            // For canonical form, we write as a map
            buf.extend_from_slice(b"{\"cnp0\":\"fixed\",\"value\":");
            buf.extend_from_slice(value.to_string().as_bytes());
            buf.push(b'}');
        }
    }
}

fn write_string(buf: &mut Vec<u8>, s: &str) {
    buf.push(b'"');
    for c in s.chars() {
        match c {
            '"' => buf.extend_from_slice(b"\\\""),
            '\\' => buf.extend_from_slice(b"\\\\"),
            '\b' => buf.extend_from_slice(b"\\b"),
            '\f' => buf.extend_from_slice(b"\\f"),
            '\n' => buf.extend_from_slice(b"\\n"),
            '\r' => buf.extend_from_slice(b"\\r"),
            '\t' => buf.extend_from_slice(b"\\t"),
            c if c < 0x20 => {
                let hex = format!("\\u{:04x}", c as u32);
                buf.extend_from_slice(hex.as_bytes());
            }
            c => buf.extend_from_slice(c.to_string().as_bytes()),
        }
    }
    buf.push(b'"');
}

    use std::collections::hash_map::DefaultHasher;
    use std::hash::{Hash, Hasher};

    // This is a placeholder for now - the real implementation would be

}
