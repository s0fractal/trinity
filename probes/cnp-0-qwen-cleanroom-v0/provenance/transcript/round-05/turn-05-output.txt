struct Parser<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> Parser<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Parser { bytes, pos: 0 }
    }

    fn peek(&self) -> Option<u8> {
        self.bytes.get(self.pos).copied()
    }

    fn consume(&mut self) -> Option<u8> {
        let byte = self.peek();
        self.pos += 1;
        byte
    }

    fn expect(&mut self, expected: u8) -> Result<(), Error> {
        match self.consume() {
            Some(b) if b == expected => Ok(()),
            _ => Err(Error::Syntax),
        }
    }

    fn parse_string(&mut self) -> Result<String, Error> {
        self.expect(b'"')?;
        let start = self.pos;
        while let Some(byte) = self.consume() {
            match byte {
                b'\\' => {
                    if let Some(escaped) = self.consume() {
                        match escaped {
                            b'"' | b'\\' | b'/' | b'b' | b'f' | b'n' | b'r' | b't' => continue,
                            b'u' => {
                                // Parse Unicode escape
                                for _ in 0..4 {
                                    if let Some(hex) = self.consume() {
                                        if !hex.is_ascii_hexdigit() {
                                            return Err(Error::MalformedEscape);
                                        }
                                    } else {
                                        return Err(Error::MalformedEscape);
                                    }
                                }
                            }
                            _ => return Err(Error::MalformedEscape),
                        }
                    } else {
                        return Err(Error::Syntax);
                    }
                }
                b'"' => break,
                _ => continue,
            }
        }
        let end = self.pos - 1; // Exclude the closing quote
        let s = std::str::from_utf8(&self.bytes[start..end])
            .map_err(|_| Error::InvalidUtf8)?;
        Ok(s.to_string())
    }

    fn parse_number(&mut self) -> Result<Value, Error> {
        let start = self.pos;
        let mut is_float = false;

        while let Some(byte) = self.peek() {
            match byte {
                b'0'..=b'9' | b'-' | b'.' | b'e' | b'E' => {
                    if byte == b'.' || byte == b'e' || byte == b'E' {
                        is_float = true;
                    }
                    self.consume();
                }
                _ => break,
            }
        }

        let num_str = std::str::from_utf8(&self.bytes[start..self.pos])
            .map_err(|_| Error::InvalidUtf8)?;

        if is_float {
            Err(Error::NumberNotCnp0Integer)
        } else {
            let n = num_str.parse::<i64>().map_err(|_| Error::IntegerOutOfRange)?;
            Ok(Value::Integer(n))
        }
    }

    fn parse_value(&mut self) -> Result<Value, Error> {
        match self.peek() {
            Some(b'{') => self.parse_object(),
            Some(b'[') => self.parse_array(),
            Some(b'"') => {
                let s = self.parse_string()?;
                Ok(Value::String(s))
            }
            Some(b't') => {
                // Parse true
                self.expect(b't')?;
                self.expect(b'r')?;
                self.expect(b'u')?;
                self.expect(b'e')?;
                Ok(Value::Bool(true))
            }
            Some(b'f') => {
                // Parse false
                self.expect(b'f')?;
                self.expect(b'a')?;
                self.expect(b'l')?;
                self.expect(b's')?;
                self.expect(b'e')?;
                Ok(Value::Bool(false))
            }
            Some(b'n') => {
                // Parse null
                self.expect(b'n')?;
                self.expect(b'u')?;
                self.expect(b'l')?;
                self.expect(b'l')?;
                Ok(Value::Null)
            }
            Some(b'0'..=b'9') | Some(b'-') => self.parse_number(),
            _ => Err(Error::Syntax),
        }
    }

    fn parse_array(&mut self) -> Result<Value, Error> {
        self.expect(b'[')?;
        let mut values = Vec::new();
        loop {
            match self.peek() {
                Some(b']') => {
                    self.consume();
                    break;
                }
                Some(_) => {
                    values.push(self.parse_value()?);
                    match self.peek() {
                        Some(b',') => {
                            self.consume();
                        }
                        Some(b']') => continue,
                        _ => return Err(Error::Syntax),
                    }
                }
                None => return Err(Error::Syntax),
            }
        }
        Ok(Value::Array(values))
    }

    fn parse_object(&mut self) -> Result<Value, Error> {
        self.expect(b'{')?;
        let mut map = HashMap::new();
        loop {
            match self.peek() {
                Some(b'}') => {
                    self.consume();
                    break;
                }
                Some(b'"') => {
                    let key = self.parse_string()?;
                    self.expect(b':')?;
                    let value = self.parse_value()?;
                    if map.contains_key(&key) {
                        return Err(Error::DuplicateMemberName);
                    }
                    map.insert(key, value);
                    match self.peek() {
                        Some(b',') => {
                            self.consume();
                        }
                        Some(b'}') => continue,
                        _ => return Err(Error::Syntax),
                    }
                }
                None => return Err(Error::Syntax),
                _ => return Err(Error::Syntax),
            }
        }
        Ok(Value::Object(map))
    }

    fn parse(&mut self) -> Result<Value, Error> {
        let value = self.parse_value()?;
        // Ensure there are no trailing bytes
        if self.pos < self.bytes.len() {
            return Err(Error::TrailingBytes);
        }
        Ok(value)
    }
}

// Simple hex decode function
fn hex_decode(hex: &str) -> Result<Vec<u8>, Error> {
    if hex.len() % 2 != 0 {
        return Err(Error::BytesHexInvalid);
    }
    
    let mut bytes = Vec::new();
    for chunk in hex.chars().collect::<Vec<char>>().chunks(2) {
        let byte_str = [chunk[0], chunk[1]].iter().collect::<String>();
        match u8::from_str_radix(&byte_str, 16) {
            Ok(byte) => bytes.push(byte),
            Err(_) => return Err(Error::BytesHexInvalid),
        }
    }
    Ok(bytes)
}

// GCD function for reducing ratios
fn gcd(a: i64, b: i64) -> i64 {
    let mut a = a.abs();
    let mut b = b.abs();
    while b != 0 {
        let temp = b;
        b = a % b;
        a = temp;
    }
    a
}
