fn process_line(line: &str) -> serde_json::Value {

        }),

            "id": id,

        }),

fn encode(json_value: &serde_json::Value, id: String) -> serde_json::Value {
    // First check that the top-level value has the required profile identifiers
    let root_obj = match json_value.as_object() {
        Some(obj) => obj,
        None => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "syntax"
        }),
    };
    
    if !root_obj.contains_key("canonical_encoding") || !root_obj.contains_key("numeric_profile") {
        return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "profile-identifier-invalid"
        });
    }
    
    let canonical_encoding = root_obj.get("canonical_encoding").and_then(|v| v.as_str());
    let numeric_profile = root_obj.get("numeric_profile").and_then(|v| v.as_str());
    
    if canonical_encoding != Some("hsp-jcs@v0") || numeric_profile != Some("cnp-0") {
        return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "profile-identifier-invalid"
        });
    }
    
    // Convert to our internal representation
    let value = match convert_to_value(json_value) {
        Ok(v) => v,
        Err(e) => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": String::from(e)
        }),
    };
    
    // Canonicalize the value
    let canonical_value = canonicalize_value(&value);
    
    // Serialize back to bytes
    let canonical_bytes = match serialize_to_bytes(&canonical_value) {
        Ok(bytes) => bytes,
        Err(e) => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": String::from(e)
        }),
    };
    
    // Compute SHA-256
    let sha256 = sha256(&canonical_bytes);
    
    // Convert bytes to hex
    let canonical_hex = hex::encode(&canonical_bytes);
    
    serde_json::json!({
        "id": id,
        "ok": true,
        "canonical_hex": canonical_hex,
        "sha256": sha256
    })
}

fn convert_to_value(json_value: &serde_json::Value) -> Result<Value, ParseError> {
    match json_value {
        serde_json::Value::Null => Ok(Value::Null),
        serde_json::Value::Bool(b) => Ok(Value::Bool(*b)),
        serde_json::Value::String(s) => Ok(Value::String(s.clone())),
        serde_json::Value::Array(a) => {
            let mut result = Vec::with_capacity(a.len());
            for item in a {
                result.push(convert_to_value(item)?);
            }
            Ok(Value::Array(result))
        },
        serde_json::Value::Object(o) => {
            // Check if this is a tagged form
            if let Some(tag) = o.get("cnp0") {
                if let Some(tag_str) = tag.as_str() {
                    match tag_str {
                        "bytes" => {
                            if let Some(hex) = o.get("hex").and_then(|v| v.as_str()) {
                                // Validate hex
                                if hex.len() % 2 != 0 {
                                    return Err(ParseError::BytesHexInvalid);
                                }
                                
                                let mut bytes = Vec::with_capacity(hex.len() / 2);
                                for chunk in hex.chars().collect::<Vec<char>>().chunks(2) {
                                    let byte_str: String = chunk.iter().collect();
                                    match u8::from_str_radix(&byte_str, 16) {
                                        Ok(byte) => bytes.push(byte),
                                        Err(_) => return Err(ParseError::BytesHexInvalid),
                                    }
                                }
                                Ok(Value::Bytes(bytes))
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        "ratio" => {
                            let num = o.get("num").and_then(|v| v.as_i64());
                            let den = o.get("den").and_then(|v| v.as_i64());
                            
                            if let (Some(n), Some(d)) = (num, den) {
                                // Check for zero
                                if n == 0 && d == 1 {
                                    // Canonical zero - OK
                                } else if n == 0 {
                                    // Zero in non-canonical form
                                    return Err(ParseError::RatioZeroNotCanonical);
                                } else if d <= 0 {
                                    return Err(ParseError::RatioNonPositiveDenominator);
                                }
                                
                                // Reduce the fraction
                                let gcd_val = gcd(n.abs(), d.abs());
                                let reduced_num = n / gcd_val;
                                let reduced_den = d / gcd_val;
                                
                                // Check if already canonical (reduced)
                                if reduced_num != n || reduced_den != d {
                                    return Err(ParseError::RatioNotReduced);
                                }
                                
                                Ok(Value::Ratio { num: reduced_num, den: reduced_den })
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        "fixed" => {
                            let value = o.get("value").and_then(|v| v.as_i64());
                            if let Some(v) = value {
                                Ok(Value::Fixed { value: v })
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        _ => Err(ParseError::TaggedFormInvalid),
                    }
                } else {
                    // Not a string tag - treat as regular object
                    let mut map = HashMap::new();
                    for (key, value) in o {
                        map.insert(key.clone(), convert_to_value(value)?);
                    }
                    Ok(Value::Object(map))
                }
            } else {
                // Regular object
                let mut map = HashMap::new();
                for (key, value) in o {
                    map.insert(key.clone(), convert_to_value(value)?);
                }
                Ok(Value::Object(map))
            }
        },
        serde_json::Value::Number(n) => {
            if let Some(i) = n.as_i64() {
                // Check range
                if i < MIN_INT || i > MAX_INT {
                    return Err(ParseError::IntegerOutOfRange);
                }
                Ok(Value::Integer(i))
            } else if let Some(f) = n.as_f64() {
                // Reject floats
                if f.is_nan() || f.is_infinite() {
                    return Err(ParseError::NumberNotCnp0Integer);
                }
                // Check for -0.0
                if f == 0.0 && f.to_bits() & 0x8000000000000000 != 0 {
                    return Err(ParseError::SignedZero);
                }
                Err(ParseError::NumberNotCnp0Integer)
            } else {
                // Not an integer or float - this shouldn't happen in valid JSON
                Err(ParseError::NumberNotCnp0Integer)
            }
        },
    }
}

fn canonicalize_value(value: &Value) -> Value {
    match value {
        Value::Null => Value::Null,
        Value::Bool(b) => Value::Bool(*b),
        Value::String(s) => Value::String(s.clone()),
        Value::Array(a) => {
            let mut result = Vec::with_capacity(a.len());
            for item in a {
                result.push(canonicalize_value(item));
            }
            Value::Array(result)
        },
        Value::Object(o) => {
            // Sort keys and canonicalize values
            let mut sorted_map: Vec<(&String, &Value)> = o.iter().collect();
            sorted_map.sort_by(|a, b| a.0.cmp(b.0));
            
            let mut map = HashMap::new();
            for (key, val) in sorted_map {
                map.insert(key.clone(), canonicalize_value(val));
            }
            Value::Object(map)
        },
        Value::Integer(i) => Value::Integer(*i),
        Value::Float(f) => Value::Float(*f), // This should never be reached
        Value::Bytes(b) => Value::Bytes(b.clone()),
        Value::Ratio { num, den } => Value::Ratio { num: *num, den: *den },
        Value::Fixed { value } => Value::Fixed { value: *value },
    }
}

fn serialize_to_bytes(value: &Value) -> Result<Vec<u8>, ParseError> {
    let json_value = match value {
        Value::Null => serde_json::Value::Null,
        Value::Bool(b) => serde_json::Value::Bool(*b),
        Value::String(s) => serde_json::Value::String(s.clone()),
        Value::Array(a) => {
            let mut result = Vec::with_capacity(a.len());
            for item in a {
                result.push(serialize_to_bytes(item)?);
            }
            serde_json::Value::Array(result)
        },
        Value::Object(o) => {
            // Convert to JSON value
            let mut map = serde_json::Map::new();
            for (key, val) in o {
                let json_val = serialize_to_bytes(val)?;
                // Reconstruct the value from bytes
                let parsed: serde_json::Value = serde_json::from_slice(&json_val).unwrap(); 
                map.insert(key.clone(), parsed);
            }
            serde_json::Value::Object(map)
        },
        Value::Integer(i) => serde_json::Value::Number(serde_json::Number::from(*i)),
        Value::Float(f) => serde_json::Value::Number(serde_json::Number::from_f64(*f).unwrap_or(0.0)),
        Value::Bytes(b) => {
            let hex = hex::encode(b);
            serde_json::json!({"cnp0": "bytes", "hex": hex})
        },
        Value::Ratio { num, den } => {
            serde_json::json!({"cnp0": "ratio", "num": num, "den": den})
        },
        Value::Fixed { value } => {
            serde_json::json!({"cnp0": "fixed", "value": value})
        },
    };
    
    // Serialize to bytes
    let serialized = serde_json::to_vec(&json_value).unwrap();
    Ok(serialized)
}

fn gcd(a: i64, b: i64) -> i64 {
    if b == 0 {
        a.abs()
    } else {
        gcd(b, a % b)
    }
}

fn process_line(line: &str) -> serde_json::Value {
    let parsed: serde_json::Value = match serde_json::from_str(line) {
        Ok(v) => v,
        Err(_) => return serde_json::json!({
            "id": "unknown",
            "ok": false,
            "category": "syntax"
        }),
    };
    
    let id = parsed["id"].as_str().unwrap_or("unknown");
    let raw_hex = parsed["raw_hex"].as_str().unwrap_or("");
    
    // Decode hex
    let bytes = match hex::decode(raw_hex) {
        Ok(b) => b,
        Err(_) => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "syntax"
        }),

        "verify" => verify(&bytes, id.to_string()),

            "id": id,
            "ok": false,

    }
}

fn encode(bytes: &[u8], id: String) -> serde_json::Value {
    // Parse the bytes as JSON
    let json_value: serde_json::Value = match serde_json::from_slice(bytes) {
        Ok(v) => v,
        Err(_) => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "syntax"
        }),
    };
    
    // Check that it's a valid JSON object with required keys
    let root_obj = match json_value.as_object() {
        Some(obj) => obj,
        None => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "syntax"
        }),
    };
    
    if !root_obj.contains_key("canonical_encoding") || !root_obj.contains_key("numeric_profile") {
        return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "profile-identifier-invalid"
        });
    }
    
    let canonical_encoding = root_obj.get("canonical_encoding").and_then(|v| v.as_str());
    let numeric_profile = root_obj.get("numeric_profile").and_then(|v| v.as_str());
    
    if canonical_encoding != Some("hsp-jcs@v0") || numeric_profile != Some("cnp-0") {
        return serde_json::json!({
            "id": id,
            "ok": false,
            "category": "profile-identifier-invalid"
        });
    }
    
    // Convert to our internal representation
    let value = match convert_to_value(&json_value) {
        Ok(v) => v,
        Err(e) => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": String::from(e)
        }),
    };
    
    // Canonicalize the value
    let canonical_value = canonicalize_value(&value);
    
    // Serialize back to bytes
    let canonical_bytes = match serialize_to_bytes(&canonical_value) {
        Ok(bytes) => bytes,
        Err(e) => return serde_json::json!({
            "id": id,
            "ok": false,
            "category": String::from(e)
        }),
    };
    
    // Compute SHA-256
    let sha256 = sha256(&canonical_bytes);
    
    // Convert bytes to hex
    let canonical_hex = hex::encode(&canonical_bytes);
    
    serde_json::json!({
        "id": id,
        "ok": true,

    })
}

fn convert_to_value(json_value: &serde_json::Value) -> Result<Value, ParseError> {
    match json_value {
        serde_json::Value::Null => Ok(Value::Null),
        serde_json::Value::Bool(b) => Ok(Value::Bool(*b)),
        serde_json::Value::String(s) => Ok(Value::String(s.clone())),
        serde_json::Value::Array(a) => {
            let mut result = Vec::with_capacity(a.len());
            for item in a {
                result.push(convert_to_value(item)?);
            }
            Ok(Value::Array(result))
        },
        serde_json::Value::Object(o) => {
            // Check if this is a tagged form
            if let Some(tag) = o.get("cnp0") {
                if let Some(tag_str) = tag.as_str() {
                    match tag_str {
                        "bytes" => {
                            if let Some(hex) = o.get("hex").and_then(|v| v.as_str()) {
                                // Validate hex
                                if hex.len() % 2 != 0 {
                                    return Err(ParseError::BytesHexInvalid);
                                }
                                
                                let mut bytes = Vec::with_capacity(hex.len() / 2);
                                for chunk in hex.chars().collect::<Vec<char>>().chunks(2) {
                                    let byte_str: String = chunk.iter().collect();
                                    match u8::from_str_radix(&byte_str, 16) {
                                        Ok(byte) => bytes.push(byte),
                                        Err(_) => return Err(ParseError::BytesHexInvalid),
                                    }
                                }
                                Ok(Value::Bytes(bytes))
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        "ratio" => {
                            let num = o.get("num").and_then(|v| v.as_i64());
                            let den = o.get("den").and_then(|v| v.as_i64());
                            
                            if let (Some(n), Some(d)) = (num, den) {
                                // Check for zero
                                if n == 0 && d == 1 {
                                    // Canonical zero - OK
                                } else if n == 0 {
                                    // Zero in non-canonical form
                                    return Err(ParseError::RatioZeroNotCanonical);
                                } else if d <= 0 {
                                    return Err(ParseError::RatioNonPositiveDenominator);
                                }
                                
                                // Reduce the fraction
                                let gcd_val = gcd(n.abs(), d.abs());
                                let reduced_num = n / gcd_val;
                                let reduced_den = d / gcd_val;
                                
                                // Check if already canonical (reduced)
                                if reduced_num != n || reduced_den != d {
                                    return Err(ParseError::RatioNotReduced);
                                }
                                
                                Ok(Value::Ratio { num: reduced_num, den: reduced_den })
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        "fixed" => {
                            let value = o.get("value").and_then(|v| v.as_i64());
                            if let Some(v) = value {
                                Ok(Value::Fixed { value: v })
                            } else {
                                Err(ParseError::TaggedFormInvalid)
                            }
                        },
                        _ => Err(ParseError::TaggedFormInvalid),
                    }
                } else {
                    // Not a string tag - treat as regular object
                    let mut map = HashMap::new();
                    for (key, value) in o {
                        map.insert(key.clone(), convert_to_value(value)?);
                    }
                    Ok(Value::Object(map))
                }
            } else {
                // Regular object
                let mut map = HashMap::new();
                for (key, value) in o {
                    map.insert(key.clone(), convert_to_value(value)?);
                }
                Ok(Value::Object(map))
            }
        },
        serde_json::Value::Number(n) => {
            if let Some(i) = n.as_i64() {
                // Check range
                if i < MIN_INT || i > MAX_INT {
                    return Err(ParseError::IntegerOutOfRange);
                }
                Ok(Value::Integer(i))
            } else if let Some(f) = n.as_f64() {
                // Reject floats
                if f.is_nan() || f.is_infinite() {
                    return Err(ParseError::NumberNotCnp0Integer);
                }
                // Check for -0.0
                if f == 0.0 && f.to_bits() & 0x8000000000000000 != 0 {
                    return Err(ParseError::SignedZero);
                }
                Err(ParseError::NumberNotCnp0Integer)
            } else {
                // Not an integer or float - this shouldn't happen in valid JSON
                Err(ParseError::NumberNotCnp0Integer)
            }
        },
    }
}

fn canonicalize_value(value: &Value) -> Value {
    match value {
        Value::Null => Value::Null,
        Value::Bool(b) => Value::Bool(*b),
        Value::String(s) => Value::String(s.clone()),
        Value::Array(a) => {
            let mut result = Vec::with_capacity(a.len());
            for item in a {
                result.push(canonicalize_value(item));
            }
            Value::Array(result)
        },
        Value::Object(o) => {
            // Sort keys and canonicalize values
            let mut sorted_map: Vec<(&String, &Value)> = o.iter().collect();
            sorted_map.sort_by(|a, b| a.0.cmp(b.0));
            
            let mut map = HashMap::new();
            for (key, val) in sorted_map {
                map.insert(key.clone(), canonicalize_value(val));
            }
            Value::Object(map)
        },
        Value::Integer(i) => Value::Integer(*i),
        Value::Float(f) => Value::Float(*f), // This should never be reached
        Value::Bytes(b) => Value::Bytes(b.clone()),
        Value::Ratio { num, den } => Value::Ratio { num: *num, den: *den },
        Value::Fixed { value } => Value::Fixed { value: *value },
    }
}

fn serialize_to_bytes(value: &Value) -> Result<Vec<u8>, ParseError> {
    // Serialize to JSON bytes - we'll use serde_json for this part
    let json_value = match value {
        Value::Null => serde_json::Value::Null,
        Value::Bool(b) => serde_json::Value::Bool(*b),
        Value::String(s) => serde_json::Value::String(s.clone()),
        Value::Array(a) => {
            let mut result = Vec::with_capacity(a.len());
            for item in a {
                result.push(serialize_to_bytes(item)?);
            }
            serde_json::Value::Array(result)
        },
        Value::Object(o) => {
            // Convert to JSON value - we'll use the canonical ordering
            let mut map = serde_json::Map::new();
            let mut sorted_map: Vec<(&String, &Value)> = o.iter().collect();
            sorted_map.sort_by(|a, b| a.0.cmp(b.0));
            
            for (key, val) in sorted_map {
                let json_val = serialize_to_bytes(val)?;
                // Parse the value to a serde_json::Value so we can insert it
                let parsed: serde_json::Value = serde_json::from_slice(&json_val).unwrap();
                map.insert(key.clone(), parsed);
            }
            serde_json::Value::Object(map)
        },
        Value::Integer(i) => serde_json::Value::Number(serde_json::Number::from(*i)),
        Value::Float(f) => serde_json::Value::Number(serde_json::Number::from_f64(*f).unwrap_or(0.0)),

        Value::Fixed { value } => {

        },
    };
    
    // Serialize to bytes
    let serialized = serde_json::to_vec(&json_value).unwrap();
    Ok(serialized)
}
