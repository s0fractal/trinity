fn parse_value(bytes: &[u8]) -> Result<Value, ParseError> {
    let s = str::from_utf8(bytes)
        .map_err(|_| ParseError::InvalidUtf8)?;
    
    // Check if it's valid JSON first
    let v: serde_json::Value = serde_json::from_str(s)

fn convert_value(v: &serde_json::Value) -> Result<Value, ParseError> {
    match v {
        serde_json::Value::Null => Ok(Value::Null),
        serde_json::Value::Bool(b) => Ok(Value::Bool(*b)),
        serde_json::Value::String(s) => {
            // Check for unpaired surrogates - this is a simplified check
            if s.chars().any(|c| (0xD800..=0xDFFF).contains(&(c as u32))) {
                return Err(ParseError::UnpairedSurrogate);
            }
            Ok(Value::String(s.clone()))
        },
        serde_json::Value::Number(n) => {
            if let Some(i) = n.as_i64() {
                // Check range
                if i < -(2i64.pow(53) - 1) || i > 2i64.pow(53) - 1 {
                    return Err(ParseError::IntegerOutOfRange);
                }
                Ok(Value::Number(i))
            } else {
                // Not an integer, so not allowed in CNP-0
                Err(ParseError::NumberNotCnp0Integer)
            }
        },
        serde_json::Value::Array(arr) => {
            let mut result = Vec::new();
            for item in arr {
                result.push(convert_value(item)?);
            }
            Ok(Value::Array(result))
        },
        serde_json::Value::Object(map) => {
            // Check for duplicate keys
            let mut seen_keys = std::collections::HashSet::new();
            for key in map.keys() {
                if !seen_keys.insert(key) {
                    return Err(ParseError::DuplicateMemberName);
                }
            }
            
            let mut result_map = HashMap::new();
            for (k, v) in map {
                result_map.insert(k.clone(), convert_value(v)?);
            }
            Ok(Value::Object(result_map))
        }
    }
}

fn encode_value(val: &Value) -> Vec<u8> {
    match val {
        Value::Null => b"null".to_vec(),
        Value::Bool(b) => if *b { b"true".to_vec() } else { b"false".to_vec() },
        Value::String(s) => {
            let mut result = Vec::new();
            result.push(b'"');
            for c in s.chars() {
                match c {
                    '"' => result.extend_from_slice(br#"\"\""#),
                    '\\' => result.extend_from_slice(br#"\\\\"#),
                    '\u{08}' => result.extend_from_slice(b"\\b"),
                    '\u{0c}' => result.extend_from_slice(b"\\f"),
                    '\n' => result.extend_from_slice(b"\\n"),
                    '\r' => result.extend_from_slice(b"\\r"),
                    '\t' => result.extend_from_slice(b"\\t"),
                    c if c < 0x20 => {
                        // Control character
                        let hex = format!("\\u{:04x}", c as u32);
                        result.extend_from_slice(hex.as_bytes());
                    }
                    c => result.extend_from_slice(c.to_string().as_bytes()),
                }
            }
            result.push(b'"');
            result
        },
        Value::Number(n) => n.to_string().into_bytes().collect::<Vec<_>>(),
        Value::Array(arr) => {
            let mut result = Vec::new();
            result.push(b'[');
            for (i, item) in arr.iter().enumerate() {
                if i > 0 {
                    result.push(b',');
                }
                result.extend_from_slice(&encode_value(item));
            }
            result.push(b']');
            result
        },
        Value::Object(map) => {
            // Canonical form: keys must be sorted and unique
            let mut pairs: Vec<(&String, &Value)> = map.iter().collect();
            pairs.sort_by(|a, b| a.0.cmp(b.0));
            
            let mut result = Vec::new();
            result.push(b'{');
            for (i, (key, value)) in pairs.iter().enumerate() {
                if i > 0 {
                    result.push(b',');
                }
                result.extend_from_slice(&encode_value(&Value::String((*key).clone())));
                result.push(b':');
                result.extend_from_slice(&encode_value(value));
            }
            result.push(b'}');
            result
        },
        Value::Bytes(_) => b"\"bytes\"".to_vec(),
        Value::Ratio { num, den } => {
            format!("{{\"cnp0\":\"ratio\",\"num\":{},\"den\":{}}}", num, den).into_bytes().collect::<Vec<_>>()
        },
        Value::Fixed { value } => {
            format!("{{\"cnp0\":\"fixed\",\"value\":{}}}", value).into_bytes().collect::<Vec<_>>()
        }
    }
}

fn hex_decode(hex: &str) -> Result<Vec<u8>, Box<dyn std::error::Error>> {
    let mut bytes = Vec::new();
    let mut chars = hex.chars().peekable();
    
    while let (Some(h), Some(l)) = (chars.next(), chars.next()) {
        if let (Some(high), Some(low)) = (hex_char_to_u8(h), hex_char_to_u8(l)) {
            bytes.push((high << 4) | low);
        } else {
            return Err("Invalid hex string".into());
        }
    }
    
    if chars.next().is_some() {
        return Err("Odd-length hex string".into());
    }
    
    Ok(bytes)
}

fn hex_char_to_u8(c: char) -> Option<u8> {
    match c {
        '0'..='9' => Some(c as u8 - b'0'),
        'a'..='f' => Some(c as u8 - b'a' + 10),
        'A'..='F' => Some(c as u8 - b'A' + 10),
        _ => None,
    }
}

fn hex_encode(bytes: &[u8]) -> String {
    let mut hex = String::new();
    for &b in bytes {
        hex.push_str(&format!("{:02x}", b));
    }
    hex
}

}

fn hex_decode(hex: &str) -> Result<Vec<u8>, Box<dyn std::error::Error>> {
    let mut bytes = Vec::new();
    let mut chars = hex.chars().peekable();
    
    while let (Some(h), Some(l)) = (chars.next(), chars.next()) {
        if let (Some(high), Some(low)) = (hex_char_to_u8(h), hex_char_to_u8(l)) {
            bytes.push((high << 4) | low);
        } else {
            return Err("Invalid hex string".into());
        }
    }
    
    if chars.next().is_some() {
        return Err("Odd-length hex string".into());
    }
    
    Ok(bytes)
}

fn hex_char_to_u8(c: char) -> Option<u8> {
    match c {
        '0'..='9' => Some(c as u8 - b'0'),
        'a'..='f' => Some(c as u8 - b'a' + 10),
        'A'..='F' => Some(c as u8 - b'A' + 10),
        _ => None,
    }
}

fn hex_encode(bytes: &[u8]) -> String {
    let mut hex = String::new();
    for &b in bytes {
        hex.push_str(&format!("{:02x}", b));
    }
    hex
}
