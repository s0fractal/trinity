fn hex_decode(hex: &str) -> Result<Vec<u8>, Box<dyn std::error::Error>> {
    let mut bytes = Vec::new();
    let mut chars = hex.chars().peekable();
    
    while let (Some(h), Some(l)) = (chars.next(), chars.next()) {
        if let (Some(high), Some(low)) = (hex_char_to_u8(h), hex_char_to_u8(l)) {
            bytes.push((high << 4) | low);
        } else {
            return Err("Invalid hex string".into());
        }
    }
    
    if chars.next().is_some() {
        return Err("Odd-length hex string".into());
    }
    
    Ok(bytes)
}

fn hex_char_to_u8(c: char) -> Option<u8> {
    match c {
        '0'..='9' => Some(c as u8 - b'0'),
        'a'..='f' => Some(c as u8 - b'a' + 10),
        'A'..='F' => Some(c as u8 - b'A' + 10),
        _ => None,
    }
}

fn parse_object(s: &str) -> Result<JsonValue, ParseError> {
    // Find the start and end of the object
    let trimmed = s.trim_start();
    if !trimmed.starts_with('{') {
        return Err(ParseError::Syntax);
    }
    
    let mut members = Vec::new();
    let mut i = 1; // skip '{'
    let chars: Vec<char> = trimmed.chars().collect();
    
    while i < chars.len() {
        // Skip whitespace
        while i < chars.len() && chars[i].is_whitespace() {
            i += 1;
        }
        
        if i >= chars.len() {
            break;
        }
        
        // Check for end of object
        if chars[i] == '}' {
            break;
        }
        
        // Parse key
        if chars[i] != '"' {
            return Err(ParseError::Syntax);
        }
        i += 1;
        let start_key = i;
        while i < chars.len() && chars[i] != '"' {
            i += 1;
        }
        if i >= chars.len() {
            return Err(ParseError::Syntax);
        }
        let key = &trimmed[start_key..i];
        i += 1; // skip '"'
        
        // Skip whitespace
        while i < chars.len() && chars[i].is_whitespace() {
            i += 1;
        }
        
        if i >= chars.len() || chars[i] != ':' {
            return Err(ParseError::Syntax);
        }
        i += 1; // skip ':'
        
        // Skip whitespace
        while i < chars.len() && chars[i].is_whitespace() {
            i += 1;
        }
        
        // Parse value
        let value = parse_value(&chars, &mut i)?;
        members.push((key.to_string(), value));
        
        // Skip whitespace
        while i < chars.len() && chars[i].is_whitespace() {
            i += 1;
        }
        
        if i >= chars.len() {
            break;
        }
        
        // Check for comma or end of object
        if chars[i] == ',' {
            i += 1;
        } else if chars[i] == '}' {
            break;
        } else {
            return Err(ParseError::Syntax);
        }
    }
    
    Ok(JsonValue::Object(members))
}

fn parse_value(chars: &[char], i: &mut usize) -> Result<JsonValue, ParseError> {
    // Skip whitespace
    while *i < chars.len() && chars[*i].is_whitespace() {
        *i += 1;
    }
    
    if *i >= chars.len() {
        return Err(ParseError::Syntax);
    }
    
    match chars[*i] {
        '"' => {
            *i += 1;
            let start = *i;
            while *i < chars.len() && chars[*i] != '"' {
                *i += 1;
            }
            if *i >= chars.len() {
                return Err(ParseError::Syntax);
            }
            let value = &chars[start..*i].iter().collect::<String>();
            *i += 1; // skip '"'
            Ok(JsonValue::String(value.to_string()))
        }
        '0'..='9' | '-' => {
            // Parse number
            let start = *i;
            if chars[*i] == '-' {
                *i += 1;
            }
            while *i < chars.len() && (chars[*i].is_ascii_digit() || chars[*i] == '.') {
                *i += 1;
            }
            
            // Check if it's a valid integer for our profile
            let num_str = &chars[start..*i].iter().collect::<String>();
            if num_str.contains('.') || num_str.contains('e') || num_str.contains('E') {
                return Err(ParseError::NumberNotCnp0Integer);
            }
            
            // Convert to i64
            match num_str.parse::<i64>() {
                Ok(num) => {
                    if num < -(2_i64.pow(53) - 1) || num > (2_i64.pow(53) - 1) {
                        Err(ParseError::IntegerOutOfRange)
                    } else {
                        Ok(JsonValue::Number(num))
                    }
                }
                Err(_) => Err(ParseError::NumberNotCnp0Integer),
            }
        }
        'n' => {
            // null
            if *i + 3 < chars.len() && chars[*i..=*i+3] == ['n', 'u', 'l', 'l'] {
                *i += 4;
                Ok(JsonValue::Null)
            } else {
                Err(ParseError::Syntax)
            }
        }
        't' => {

            if *i + 3 < chars.len() && chars[*i..=*i+3] == ['t', 'r', 'u', 'e'] {

                Ok(JsonValue::Bool(false))
            } else {
                Err(ParseError::Syntax)
            }
        }
        '{' => {
            // Object - let's parse it
            let start = *i;
            *i += 1; // skip '{'
            
            // For simplicity, just find the matching }
            let mut brace_count = 1;
            while *i < chars.len() && brace_count > 0 {
                if chars[*i] == '{' {

                } else if chars[*i] == '}' {
                    brace_count -= 1;
                }
                *i += 1;
            }
            
            // Reconstruct the object string and parse it
            let obj_str = &chars[start..*i].iter().collect::<String>();

        '[' => {

                }

            // Canonicalize nested values
            let canonicalized: Vec<(String, JsonValue)> = sorted_members

fn serialize_json(value: &JsonValue) -> Vec<u8> {

        JsonValue::Number(n) => n.to_string(),
        JsonValue::String(s) => format!("\"{}\"", s),
        JsonValue::Array(elements) => {
            let serialized: Vec<String> = elements.iter().map(|e| serialize_json(e).into_iter().map(|b| b as char).collect::<String>()).collect();
            format!("[{}]", serialized.join(","))
        }
        JsonValue::Object(members) => {
            let serialized: Vec<String> = members.iter()
                .map(|(k, v)| {
                    let value_str = serialize_json(v).into_iter().map(|b| b as char).collect::<String>();
                    format!("\"{}\":{}", k, value_str)
                })
                .collect();
            format!("{{{}}}", serialized.join(","))
        }
        JsonValue::Bytes(hex) => format!(r#"{{"cnp0":"bytes","hex":"{}"}}"#, hex),
        JsonValue::Ratio { num, den } => format!(r#"{{"cnp0":"ratio","num":{},"den":{}}}"#, num, den),
        JsonValue::Fixed(value) => format!(r#"{{"cnp0":"fixed","value":{}}}"#, value),
    };
    
    json_str.into_bytes().collect::<Vec<u8>>()
