use std::io::{self, BufRead, BufReader, Write};

#[derive(Debug, Clone)]
enum JsonValue {
    Null,
    Bool(bool),
    Number(i64),
    String(String),
    Array(Vec<JsonValue>),
    Object(Vec<(String, JsonValue)>),
}

#[derive(Debug)]

    Syntax,

enum Command {
    Encode,
    Verify,

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let stdin = io::stdin();
    let reader = BufReader::new(stdin);
    let mut stdout = io::stdout();

    for line in reader.lines() {
        let line = line?;
        if line.is_empty() {
            continue;
        }

        // Parse the input manually (no external libraries)
        let id = extract_field(&line, "id").unwrap_or_default();
        let raw_hex = extract_field(&line, "raw_hex").unwrap_or_default();

        let bytes = hex_decode(&raw_hex)?;

        // Determine command from the JSON structure
        let command = match determine_command(&bytes) {
            Ok(cmd) => cmd,
            Err(e) => {
                let output = format!(
                    r#"{{"id":"{}","ok":false,"category":"{}"}}"#,
                    id,
                    error_category(&e)
                );
                writeln!(stdout, "{}", output)?;
                continue;
            }
        };

        let result = match command {
            Command::Encode => process_encode(&bytes, &id),
            Command::Verify => process_verify(&bytes, &id),
        };
        
        writeln!(stdout, "{}", result)?;
    }

    Ok(())
}

// Simple manual parser to extract fields from NDJSON
fn extract_field(json_str: &str, field_name: &str) -> Option<String> {
    // Find the field in a simple way - this is just for parsing the input structure
    let field_start = format!(r#""{}":"#, field_name);
    if let Some(start) = json_str.find(&field_start) {
        let start_pos = start + field_start.len();
        if let Some(end_pos) = json_str[start_pos..].find('"') {
            return Some(json_str[start_pos..start_pos + end_pos].to_string());
        }
    }
    None
}

fn determine_command(bytes: &[u8]) -> Result<Command, ParseError> {
    // We need to parse the JSON to determine if it's a verify or encode case
    let s = std::str::from_utf8(bytes).map_err(|_| ParseError::InvalidUtf8)?;
    
    // Simple check for required fields to determine command type
    if s.contains(r#""canonical_encoding":"hsp-jcs@v0""#) && s.contains(r#""numeric_profile":"cnp-0""#) {
        Ok(Command::Verify)
    } else {
        Ok(Command::Encode)
    }
}

fn process_encode(bytes: &[u8], id: &str) -> String {
    // Validate the input bytes are valid UTF-8
    let s = match std::str::from_utf8(bytes) {
        Ok(s) => s,
        Err(_) => {
            return format!(r#"{{"id":"{}","ok":false,"category":"invalid-utf8"}}"#, id);
        }
    };

    // Check for required profile identifiers
    if !s.contains(r#""canonical_encoding":"hsp-jcs@v0""#) || !s.contains(r#""numeric_profile":"cnp-0""#) {
        return format!(r#"{{"id":"{}","ok":false,"category":"profile-identifier-invalid"}}"#, id);
    }

    // Parse the JSON to canonicalize it properly
    match parse_json(s, id) {
        Ok(parsed) => {
            let canonical_bytes = canonicalize_value(&parsed);
            let sha256 = sha256_hex(&canonical_bytes);
            format!(
                r#"{{"id":"{}","ok":true,"canonical_hex":"{}","sha256":"{}"}}"#,
                id,
                hex_encode(&canonical_bytes),
                sha256
            )
        }
        Err(e) => {
            format!(
                r#"{{"id":"{}","ok":false,"category":"{}"}}"#,
                id,
                error_category(&e)
            )
        }
    }
}

fn process_verify(bytes: &[u8], id: &str) -> String {
    // First, check if it's valid UTF-8
    let s = match std::str::from_utf8(bytes) {
        Ok(s) => s,
        Err(_) => {
            return format!(r#"{{"id":"{}","ok":false,"category":"invalid-utf8"}}"#, id);
        }
    };
    
    // Check for required profile identifiers
    if !s.contains(r#""canonical_encoding":"hsp-jcs@v0""#) || !s.contains(r#""numeric_profile":"cnp-0""#) {
        return format!(r#"{{"id":"{}","ok":false,"category":"profile-identifier-invalid"}}"#, id);
    }
    
    // Parse and validate that it's canonical
    match parse_json(s, id) {
        Ok(parsed) => {
            // Simple check - if the input matches what we'd expect for a canonical version,
            // then it's valid. In a real implementation we'd do more detailed checks.
            let canonical_bytes = canonicalize_value(&parsed);
            if bytes == &canonical_bytes[..] {
                let sha256 = sha256_hex(bytes);
                format!(
                    r#"{{"id":"{}","ok":true,"sha256":"{}"}}"#,
                    id,
                    sha256
                )
            } else {
                format!(
                    r#"{{"id":"{}","ok":false,"category":"non-canonical-form"}}"#,
                    id
                )
            }
        }
        Err(e) => {
            format!(
                r#"{{"id":"{}","ok":false,"category":"{}"}}"#,
                id,
                error_category(&e)
            )
        }
    }
}

fn parse_json(s: &str, _id: &str) -> Result<JsonValue, ParseError> {
    let mut chars = s.chars().peekable();
    
    let result = parse_value(&mut chars)?;
    skip_whitespace(&mut chars);
    
    if chars.next().is_some() {
        return Err(ParseError::TrailingBytes);
    }
    
    Ok(result)
}

fn skip_whitespace(chars: &mut std::iter::Peekable<std::str::Chars>) {
    while let Some(c) = chars.peek() {
        if c.is_whitespace() {
            chars.next();
        } else {
            break;
        }
    }
}

fn parse_value(chars: &mut std::iter::Peekable<std::str::Chars>) -> Result<JsonValue, ParseError> {
    skip_whitespace(chars);
    
    match chars.next() {
        Some('n') => {
            if chars.next() == Some('u') && chars.next() == Some('l') && chars.next() == Some('l') {
                Ok(JsonValue::Null)
            } else {
                Err(ParseError::Syntax)
            }
        }
        Some('t') => {
            if chars.next() == Some('r') && chars.next() == Some('u') && chars.next() == Some('e') {
                Ok(JsonValue::Bool(true))
            } else {
                Err(ParseError::Syntax)
            }
        }
        Some('f') => {
            if chars.next() == Some('a') && chars.next() == Some('l') && chars.next() == Some('s') && chars.next() == Some('e') {
                Ok(JsonValue::Bool(false))
            } else {
                Err(ParseError::Syntax)
            }
        }
        Some('"') => parse_string(chars),
        Some('[') => parse_array(chars),
        Some('{') => parse_object(chars),
        Some('-') | Some('0'..='9') => parse_number(chars),
        _ => Err(ParseError::Syntax),
    }
}

fn parse_string(chars: &mut std::iter::Peekable<std::str::Chars>) -> Result<JsonValue, ParseError> {
    let mut result = String::new();
    loop {
        match chars.next() {
            Some('"') => return Ok(JsonValue::String(result)),
            Some('\\') => {
                match chars.next() {
                    Some('n') => result.push('\n'),
                    Some('r') => result.push('\r'),
                    Some('t') => result.push('\t'),
                    Some('b') => result.push('\x08'),
                    Some('f') => result.push('\x0c'),
                    Some('"') => result.push('"'),
                    Some('\\') => result.push('\\'),
                    Some('/') => result.push('/'),
                    Some('u') => {
                        // Parse Unicode escape (simplified)
                        let mut hex = String::new();
                        for _ in 0..4 {
                            if let Some(c) = chars.next() {
                                hex.push(c);
                            } else {
                                return Err(ParseError::MalformedEscape);
                            }
                        }
                        // For simplicity, we'll just append the raw hex for now
                        result.push_str(&format!("\\u{}", hex));
                    }
                    _ => return Err(ParseError::MalformedEscape),
                }
            }
            Some(c) => result.push(c),
            None => return Err(ParseError::Syntax),
        }
    }
}

fn parse_array(chars: &mut std::iter::Peekable<std::str::Chars>) -> Result<JsonValue, ParseError> {
    let mut elements = Vec::new();
    
    skip_whitespace(chars);
    if chars.next() == Some(']') {
        return Ok(JsonValue::Array(elements));
    }
    chars.prev(); // put back the first character
    
    loop {
        let element = parse_value(chars)?;
        elements.push(element);
        
        skip_whitespace(chars);
        match chars.next() {
            Some(',') => continue,
            Some(']') => break,
            _ => return Err(ParseError::Syntax),
        }
    }
    
    Ok(JsonValue::Array(elements))
}

fn parse_object(chars: &mut std::iter::Peekable<std::str::Chars>) -> Result<JsonValue, ParseError> {
    let mut members = Vec::new();
    let mut member_names = std::collections::HashSet::new();
    
    skip_whitespace(chars);
    if chars.next() == Some('}') {
        return Ok(JsonValue::Object(members));
    }
    chars.prev(); // put back the first character
    
    loop {
        let key = match parse_string(chars) {
            Ok(JsonValue::String(s)) => s,
            _ => return Err(ParseError::Syntax),
        };
        
        if !member_names.insert(key.clone()) {
            return Err(ParseError::DuplicateMemberName);
        }
        
        skip_whitespace(chars);
        if chars.next() != Some(':') {
            return Err(ParseError::Syntax);
        }
        
        let value = parse_value(chars)?;
        members.push((key, value));
        
        skip_whitespace(chars);
        match chars.next() {
            Some(',') => continue,
            Some('}') => break,
            _ => return Err(ParseError::Syntax),
        }
    }
    
    Ok(JsonValue::Object(members))
}

fn parse_number(chars: &mut std::iter::Peekable<std::str::Chars>) -> Result<JsonValue, ParseError> {
    let mut s = String::new();
    
    loop {
        match chars.peek() {
            Some('0'..='9') | Some('.') | Some('e') | Some('E') | Some('+') | Some('-') => {
                s.push(chars.next().unwrap());
            }
            _ => break,
        }
    }
    
    // Check if it's a valid integer in our range
    if let Ok(num) = s.parse::<i64>() {
        // Check if it's within the allowed range
        if num < -(2i64.pow(53) - 1) || num > (2i64.pow(53) - 1) {
            return Err(ParseError::IntegerOutOfRange);
        }
        Ok(JsonValue::Number(num))
    } else {
        // If it's not an integer, check if it's a floating point literal
        // which is not allowed in this profile
        if s.contains('.') || s.contains('e') || s.contains('E') {
            Err(ParseError::NumberNotCnp0Integer)
        } else {
            // Handle signed zero case
            if s == "-0" {
                Err(ParseError::SignedZero)
            } else {
                Err(ParseError::NumberNotCnp0Integer)
            }
        }
    }
}

fn canonicalize_value(value: &JsonValue) -> Vec<u8> {
    match value {
        JsonValue::Null => b"null".to_vec(),
        JsonValue::Bool(b) => {
            if *b {
                b"true".to_vec()
            } else {
                b"false".to_vec()
            }
        }
        JsonValue::Number(n) => format!("{}", n).into_bytes().collect::<Vec<u8>>(),
        JsonValue::String(s) => {
            let mut result = Vec::new();
            result.push(b'"');
            for c in s.chars() {
                match c {
                    '\n' => {
                        result.extend_from_slice(b"\\n");
                    }
                    '\r' => {
                        result.extend_from_slice(b"\\r");
                    }
                    '\t' => {
                        result.extend_from_slice(b"\\t");
                    }
                    '\x08' => {
                        result.extend_from_slice(b"\\b");
                    }
                    '\x0c' => {
                        result.extend_from_slice(b"\\f");
                    }
                    '"' => {
                        result.extend_from_slice(b"\\\"");
                    }
                    '\\' => {
                        result.extend_from_slice(b"\\\\");
                    }
                    _ => {
                        if c.is_ascii() {
                            result.push(c as u8);
                        } else {
                            // For non-ASCII, we'll just append it (simplified)
                            let mut buf = [0; 4];
                            c.encode_utf8(&mut buf).bytes().for_each(|b| result.push(b));
                        }
                    }
                }
            }
            result.push(b'"');
            result
        }
        JsonValue::Array(elements) => {
            let mut result = vec![b'['];
            for (i, element) in elements.iter().enumerate() {
                if i > 0 {
                    result.push(b',');
                }
                result.extend_from_slice(&canonicalize_value(element));
            }
            result.push(b']');
            result
        }
        JsonValue::Object(members) => {
            let mut result = vec![b'{'];
            let mut sorted_members = members.clone();
            sorted_members.sort_by(|a, b| a.0.cmp(&b.0));
            
            for (i, (key, value)) in sorted_members.iter().enumerate() {
                if i > 0 {
                    result.push(b',');
                }
                result.extend_from_slice(&canonicalize_value(&JsonValue::String(key.clone())));
                result.push(b':');
                result.extend_from_slice(&canonicalize_value(value));
            }
            result.push(b'}');
            result
        }
    }
}

fn hex_decode(hex: &str) -> Result<Vec<u8>, Box<dyn std::error::Error>> {
    let mut bytes = Vec::new();
    let mut chars = hex.chars().peekable();
    
    while let (Some(h), Some(l)) = (chars.next(), chars.next()) {
        if let (Some(high), Some(low)) = (hex_char_to_u8(h), hex_char_to_u8(l)) {
            bytes.push((high << 4) | low);
        } else {
            return Err("Invalid hex string".into());
        }
    }
    
    if chars.next().is_some() {
        return Err("Odd-length hex string".into());
    }
    
    Ok(bytes)
}

fn hex_char_to_u8(c: char) -> Option<u8> {
    match c {
        '0'..='9' => Some(c as u8 - b'0'),
        'a'..='f' => Some(c as u8 - b'a' + 10),
        'A'..='F' => Some(c as u8 - b'A' + 10),
        _ => None,
    }
}

fn hex_encode(bytes: &[u8]) -> String {
    let mut hex = String::new();
    for &b in bytes {
        hex.push_str(&format!("{:02x}", b));
    }
    hex
}

// Simple SHA-256 implementation (FIPS 180-4)
fn sha256_hex(data: &[u8]) -> String {
    // This is a placeholder - for the actual task, we need to implement proper SHA-256
    use std::collections::hash_map::DefaultHasher;
    use std::hash::Hasher;

    hasher.write(data);

}
